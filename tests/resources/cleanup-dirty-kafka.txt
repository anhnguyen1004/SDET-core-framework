import time
from confluent_kafka.admin import AdminClient, NewTopic
from src.config.settings import settings

admin = AdminClient({'bootstrap.servers': settings.KAFKA_BROKER})
metadata = admin.list_topics(timeout=5)

# 1. Tìm các topic rác từ test
topics_to_delete = [t for t in metadata.topics if t.startswith('sdet.bao-cao.created.')]

# 2. Thêm dead-letter-queue và sdet.bao-cao.created để reset sạch message
topics_to_reset = ['dead-letter-queue', 'sdet.bao-cao.created']
for t in topics_to_reset:
    if t in metadata.topics:
        topics_to_delete.append(t)

print(f'Deleting {len(topics_to_delete)} topics...')
fs = admin.delete_topics(topics_to_delete)
for topic, f in fs.items():
    try:
        f.result(timeout=5)
        print(f' - Deleted: {topic}')
    except Exception as e:
        print(f' - Error deleting {topic}: {e}')

time.sleep(1)

# 3. Tạo lại các topic chính ở trạng thái hoàn toàn rỗng
new_topics = [NewTopic(t, num_partitions=1, replication_factor=1) for t in topics_to_reset]
fs_create = admin.create_topics(new_topics)
for topic, f in fs_create.items():
    try:
        f.result(timeout=5)
        print(f' - Recreated clean: {topic}')
    except Exception as e:
        print(f' - Error creating {topic}: {e}')

print('DONE! Broker is now 100% clean.')
